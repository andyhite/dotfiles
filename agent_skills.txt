# Cross-agent skills, one per line: <owner>/<repo> --skill <name>
# install.sh parses each line and runs:
#   npx skills add <owner>/<repo> --skill <name> -g -y
# -g installs globally rather than into a single project, and -y accepts the
# CLI's prompts unattended so this can run inside install.sh without a tty.
#
# The canonical copies land in ~/.agents/skills; the CLI then symlinks each
# configured agent's own skills directory into that tree, which is how omp
# (and every other agent) picks them up without a separate install per tool.
#
# ~/.agents/.skill-lock.json stays untracked — it's generated state (content
# hashes, install/update timestamps), exactly like herdr's own plugins.json
# next to herdr_plugins.txt.
herdrdev/herdr --skill herdr
vercel-labs/skills --skill find-skills
ayghri/i-have-adhd --skill i-have-adhd

# Paseo's four orchestration skills, from the same repo the daemon is built
# from. The macOS desktop app also copies these into ~/.agents/skills on its
# first GUI launch, but that hook is useless here twice over: it never fires on
# the Linux box (no desktop app at all), and on the Mac it only fires once
# someone opens the app, which `brew bundle` never does. Going through the
# manifest instead means both machines get them from `install.sh` alone, and
# they update with everything else rather than whenever the app is next
# launched.
#
# `paseo` is the one the others depend on: it documents the tool surface and the
# ~/.paseo/orchestration-preferences.json contract that decides which provider
# each delegated role gets.
getpaseo/paseo --skill paseo
getpaseo/paseo --skill paseo-advisor
getpaseo/paseo --skill paseo-committee
getpaseo/paseo --skill paseo-handoff
